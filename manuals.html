// manuals-loader.js (full) document.addEventListener('DOMContentLoaded', async () => { const titleEl = document.getElementById('manuals-title'); const gridEl = document.querySelector('.manuals-list-container'); const manualSelect = document.getElementById('manual-select'); // อ่านพารามิเตอร์ ?device=PP11 const params = new URLSearchParams(window.location.search); const areaParam = (params.get('device') || '').trim(); // ชื่อโฟลเดอร์จริงใต้ /documents // ดึงฐานข้อมูลคู่มือจาก backend let data; try { const res = await fetch('/api/manuals', { method: 'GET' }); if (!res.ok) throw new Error(`HTTP ${res.status}`); data = await res.json(); } catch (e) { console.error('Failed to load /api/manuals:', e); if (titleEl) titleEl.textContent = 'โหลดรายการคู่มือไม่สำเร็จ'; return; } // ฟังก์ชันช่วย: สร้างการ์ด const createManualCard = (areaName, file) => { const card = document.createElement('div'); card.className = 'manual-card'; // รูปปก (optional) const img = document.createElement('img'); img.alt = file.displayName || file.name; img.loading = 'lazy'; img.src = file.image || 'images/placeholder.png'; img.onerror = () => { img.src = 'images/placeholder.png'; }; img.style.maxWidth = '100%'; img.style.borderRadius = '8px'; const name = document.createElement('div'); name.className = 'manual-name'; name.textContent = file.displayName || file.name; const actions = document.createElement('div'); actions.className = 'manual-actions'; // ปุ่มเปิด PDF const openBtn = document.createElement('a'); openBtn.href = `/${file.path}`; // เช่น documents/PP11/PP11_CO_M300E.pdf openBtn.target = '_blank'; openBtn.rel = 'noopener noreferrer'; openBtn.className = 'button-link'; openBtn.textContent = 'เปิดดู PDF'; // ปุ่มคุยกับ AI (เลือกไฟล์นี้ในดรอปดาวน์) const askBtn = document.createElement('button'); askBtn.className = 'button-secondary'; askBtn.textContent = 'คุยกับ AI'; askBtn.addEventListener('click', () => { if (manualSelect) { manualSelect.value = file.name; // ⚠️ ใช้ชื่อไฟล์จริง .pdf manualSelect.dispatchEvent(new Event('change')); // เลื่อนหน้าจอไปที่กล่องแชท const chatBox = document.querySelector('.chat-container'); if (chatBox) chatBox.scrollIntoView({ behavior: 'smooth', block: 'start' }); } }); actions.appendChild(openBtn); actions.appendChild(askBtn); card.appendChild(img); card.appendChild(name); card.appendChild(actions); return card; }; // เคลียร์กริดก่อนเริ่ม if (gridEl) gridEl.innerHTML = ''; // เติมหัวข้อและเรนเดอร์การ์ด if (areaParam && data[areaParam.toLowerCase()]) { const areaObj = data[areaParam.toLowerCase()]; if (titleEl) titleEl.textContent = `คู่มือในหมวด ${areaObj.name}`; areaObj.files.forEach(file => { gridEl.appendChild(createManualCard(areaObj.name, file)); }); // เติมดรอปดาวน์เฉพาะ Area นี้ populateManualSelect([areaObj]); } else { if (titleEl) titleEl.textContent = 'รายการคู่มือทั้งหมด'; const allAreas = Object.values(data); allAreas.forEach(areaObj => { // แทรกหัวข้อย่อยของแต่ละ Area const subHeader = document.createElement('h3'); subHeader.textContent = areaObj.name; subHeader.style.margin = '16px 0 8px'; gridEl.appendChild(subHeader); const subGrid = document.createElement('div'); subGrid.className = 'manuals-subgrid'; areaObj.files.forEach(file => { subGrid.appendChild(createManualCard(areaObj.name, file)); }); gridEl.appendChild(subGrid); }); // เติมดรอปดาวน์: รวมทุก Area populateManualSelect(Object.values(data)); } // เติมตัวเลือกใน
